{{/* Wiki link render hook. migrate.py converts [[Recipe Name]] to [Recipe Name](Recipe Name) — a link where the href is the recipe title (no scheme, no slash, no anchor). This hook intercepts those links, looks up the target page by title, and emits a proper Hugo URL. Standard http/https/relative links pass through. */}} {{ $destination := .Destination }} {{ $text := .Text | safeHTML }} {{ $isWikiLink := and (not (strings.HasPrefix $destination "http")) (not (strings.HasPrefix $destination "/")) (not (strings.HasPrefix $destination "#")) (not (strings.HasPrefix $destination "mailto")) (not (strings.HasPrefix $destination "./")) (not (strings.HasPrefix $destination "../")) }} {{ if $isWikiLink }} {{ $matches := where site.RegularPages "Title" $destination }} {{ if gt (len $matches) 0 }} {{ $target := index $matches 0 }} {{ $text }} {{ else }} {{ $text }} {{ end }} {{ else }} {{ $text }} {{ end }}
