{{/* Wiki link render hook. migrate.py converts [[Recipe Name]] to [Recipe Name](Recipe Name) — a link where the href is the recipe title (no scheme, no slash, no anchor). This hook intercepts those links, looks up the target page by title, and emits a proper Hugo URL. Standard http/https/relative links pass through. */}} {{ $destination := .Destination }} {{ $text := .Text | safeHTML }} {{ $isWikiLink := and (not (strings.HasPrefix $destination "http")) (not (strings.HasPrefix $destination "/")) (not (strings.HasPrefix $destination "#")) (not (strings.HasPrefix $destination "mailto")) (not (strings.HasPrefix $destination "./")) (not (strings.HasPrefix $destination "../")) }} {{ if $isWikiLink }} {{/* Normalize curly vs. straight apostrophes before matching — real content already disagrees (content/recipes/chef-johns-hamburger-buns.md's title uses a curly ', a wiki link to it was typed with a straight '). where's exact-match can't do this, so match by hand instead. */}} {{ $normDest := replace (replace $destination "’" "'") "‘" "'" }} {{ $matches := slice }} {{ range site.RegularPages }} {{ $normTitle := replace (replace .Title "’" "'") "‘" "'" }} {{ if eq $normTitle $normDest }} {{ $matches = $matches | append . }} {{ end }} {{ end }} {{ if gt (len $matches) 0 }} {{ $target := index $matches 0 }} {{ $text }} {{ else }} {{ $text }} {{ end }} {{ else }} {{ $isExternal := strings.HasPrefix $destination "http" }} {{ $text }} {{ end }}
