{{/* Wiki link render hook. migrate.py converts [[Recipe Name]] to [Recipe Name](Recipe Name) — a link where the href is the recipe title (no scheme, no slash, no anchor). This hook intercepts those links, looks up the target page by title, and emits a proper Hugo URL. Standard http/https/relative links pass through. */}} {{ $destination := .Destination }} {{ $text := .Text | safeHTML }} {{ $isWikiLink := and (not (strings.HasPrefix $destination "http")) (not (strings.HasPrefix $destination "/")) (not (strings.HasPrefix $destination "#")) (not (strings.HasPrefix $destination "mailto")) (not (strings.HasPrefix $destination "./")) (not (strings.HasPrefix $destination "../")) }} {{ if $isWikiLink }} {{/* Normalize both sides before matching: lowercase, and collapse any run of non-alphanumeric characters (apostrophes, slashes, hyphens, ...) to a single space. This makes matching case-insensitive and punctuation- agnostic, so `[[rich veg mushroom stock]]` matches a title of "Rich veg/mushroom stock", and curly vs. straight apostrophes (content/recipes/chef-johns-hamburger-buns.md's title uses a curly ', a wiki link to it might be typed with a straight ') no longer matter — exact-match can't do either, so normalize by hand instead. */}} {{ $normDest := strings.TrimSpace (replaceRE `[^a-z0-9]+` " " (strings.ToLower $destination)) }} {{ $matches := slice }} {{ range site.RegularPages }} {{ $normTitle := strings.TrimSpace (replaceRE `[^a-z0-9]+` " " (strings.ToLower .Title)) }} {{ if eq $normTitle $normDest }} {{ $matches = $matches | append . }} {{ end }} {{ end }} {{ if gt (len $matches) 0 }} {{ $target := index $matches 0 }} {{ $text }} {{ else }} {{ $text }} {{ end }} {{ else }} {{ $isExternal := strings.HasPrefix $destination "http" }} {{ $text }} {{ end }}
